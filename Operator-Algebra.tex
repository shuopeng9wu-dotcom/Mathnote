		\begin{Def}
			\begin{itemize}
				\item An \textbf{algebra} is: A vector space $A$ with bilinear map
				\begin{gather*}
					\cdot:A\times A\rightarrow A
				\end{gather*}
				
				which satisfies: Associativity
				\begin{gather*}
					a\cdot (b\cdot c)=(a\cdot b)\cdot c
				\end{gather*}
				\item A \textbf{subalgebra} is: A vector space $B\subseteq A$ with
				\begin{gather*}
					b,b'\in B\Rightarrow bb'\in B
				\end{gather*}
				
				\item Norm on algebra
				\begin{gather*}
					\Vert ab\Vert\leq \Vert a\Vert\Vert b\Vert
				\end{gather*}
			\end{itemize}
		\end{Def}
		
		\begin{proposition}
			For a norm $\Vert-\Vert$ on algebra $A$, we obtain that
			\begin{gather*}
				\Vert ab-a'b'\Vert\leq \Vert a\Vert\Vert b-b'\Vert+\Vert a-a'\Vert\Vert b'\Vert
			\end{gather*}
		\end{proposition}
