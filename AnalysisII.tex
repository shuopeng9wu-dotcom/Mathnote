	\section{Uniform Covergent}
		Recall:
		\begin{Def}
			$\{f_n\}$ of sequence in $Func(E,\RR)$ is \textbf{uniform covergent} to $f$ if:
			\begin{gather*}
				\forall \epsilon>0\exists N(n\geq N\Rightarrow \sup_{x\in E} |f_n(x)-f(x)|<\epsilon)
			\end{gather*}
		\end{Def}
		
		\begin{proposition}
			(Cauchy) Equivalently, $\forall \epsilon>0\exists N(m,n\geq N\Rightarrow \sup_{x\in E} |f_n(x)-f_m(x)|<\epsilon)$
