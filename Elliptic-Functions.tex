		\section{Period}
			\subsection{Definitions, and Some Properties}
				\begin{Def}
					Let $f$ be a meromorphic function. Then, the \textbf{period} is $\omega\in \CC-\{0\}$ with
					\begin{gather*}
						f(z)=f(z+\omega)
					\end{gather*}
				\end{Def}
				
				It is easy to check, that 
				\begin{proposition}
					If $\omega$ is a period, so is $k\omega$. 
				\end{proposition}
