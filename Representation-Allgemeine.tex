		\section{Representations}
			\subsection{Direct Sum and Reducibility}
				\begin{Def}
					Given two representations $(\rho_1,V_1)$ and $(\rho_2,V_2)$, then, the \textbf{direct sum} of two representations is 
					\begin{gather*}
						\rho_1\oplus \rho_2:G\rightarrow End(V_1\oplus V_2)\qquad g\mapsto  (\rho_1+\rho_2)(g)
					\end{gather*}
				\end{Def}
				
				Then,
				\begin{Def}
					\begin{itemize}
						\item A representation $(\rho,V)$ is called \textbf{irreducible} if it contains no proper invariant subspaces;
						\item A representation $(\rho,V)$ is called \textbf{completely reducible} if it decomposes as a direct sum of irreducible subrepresentations;
					\end{itemize}
				\end{Def}
