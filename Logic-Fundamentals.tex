			\subsection{Mathematic Structure}
				\begin{Def}
					A \textbf{mathematic structure} consists of following elements:
					\begin{itemize}
						\item Universe $A$;
						\item Relations $R_1,\dots$;
						\item Functions $F_1,\dots$, where
						\begin{gather*}
							F_i:A^{k_i}\rightarrow A
						\end{gather*}
						\item Constants $a_1,\dots$;
					\end{itemize}
					
					We will take the following notion 
					\begin{gather*}
						\mathfrak{A}:=(A,R_1,\dots;F_1,\dots;a_1,\dots)
					\end{gather*}
				\end{Def}
				
				
				We will refer to some structures: The details will not be listed here.
				\begin{itemize}
					\item Ordering Relations: Partial Ordering;Linear Ordering; Strict Partial Ordering;
					
					Total/Connected Ordering;
					\item Algebraic: Groups, Rings and Fields;
					
					\item Universe of Partial Order Set;
				\end{itemize}
				
			\subsection{First-Order Language}
				\begin{Def}
					A \textbf{first-Order Language} $L$ is specified by: Symbols:
					\begin{itemize}
						\item \textbf{Nonlogical Symbols}
						\begin{itemize}
							\item $L$-Relation symbols $Rs_L$;
							\item $L$-Function symbols $Fs_L$;
							\item $L$-Constant Symbols $Cs_L$;
						\end{itemize}
						
						Equipped with a function
						\begin{gather*}
							v_L:Rs_L\cup Fs_L\rightarrow \omega- \{0\}
						\end{gather*}
						\item \textbf{Logical Symbols}
						\begin{itemize}
							\item Variables: A denumerable set $Vb=\{v_n|n\in \omega\}$;
							\item The equality symbols $\dot{=}$;
							\item The connectives $\neg,\vee,\wedge,\rightarrow,\leftarrow$;
							\item The quantifiers $\exists,\forall$;
						\end{itemize}
					\end{itemize}
				\end{Def}
