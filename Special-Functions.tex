		\chapter{Mollifiers}
			What is Mollifer?
			
			\begin{Def}
				Let $\varphi\in C^\infty(\RR^n)$ be a \textbf{smooth function} on $\RR^n$, $n\geq 1$. and put
				\begin{gather*}
					\varphi_\epsilon(x):=\epsilon^{-n}\varphi(\frac{x}{\epsilon})\qquad \RR\ni \epsilon>0
				\end{gather*}
				
				Then, $\varphi$ is a \textbf{mollifier} if it satisfies the following three requirements:
				\begin{itemize}
					\item It is compactly supported: The closed support 
					\begin{gather*}
						supp_C(f):=\{x\in X|f(x)\not=0\}
					\end{gather*} 
					
					is closed.  
					\item $\int_{\RR^n} \varphi(x)dx=1$;
				\end{itemize}
				
				Then,
				\begin{gather*}
					lim_{\epsilon\rightarrow 0}\varphi_\epsilon(x)=\lim_{\epsilon\rightarrow 0}\epsilon^{-n}\varphi(x/\epsilon)=\delta(x)
				\end{gather*}
				
				where: $\delta(x)$ is the \textbf{Dirac delta function}.
			\end{Def}
